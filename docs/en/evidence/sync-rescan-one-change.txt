1. One rescan and wait over 1,000 files after one of them changed

The image was built from a clean checkout of commit ce74141d. The script starts the server as
listing-bench.sh in sync-listing-attribute-cache.txt does, changes one file on the host once the
server is healthy, and then runs one rescan and one wait. One measurement, not a gate.

$ docker build --platform linux/arm64 --build-arg CARGO_BUILD_JOBS=8 -t gcs-mimic-onechange:ce74141d .
$ cat <temp>/one-change-bench.sh
#!/bin/zsh
# One measurement of `sync rescan` followed by `sync wait` after one of 1,000 files changed on a
# Docker Desktop bind mount. The start follows listing-bench.sh.
# Usage: one-change-bench.sh IMAGE RUN   (RUN names a fresh directory)
set -u
zmodload zsh/datetime
image=$1
scratch=<temp>
base=$scratch/one-change-bench/$2
name=gcs-mimic-onechange-bench

mkdir -p "$base/sync/bench-bucket"
for i in $(seq -w 1 1000); do
  printf 'object %s\n' "$i" > "$base/sync/bench-bucket/file-$i.txt"
done
echo "files: $(ls "$base/sync/bench-bucket" | wc -l)"

# Health turns healthy once the first pass has ingested and exported the tree. Reading the event
# stream blocks until Docker reports it, so nothing polls.
since=$(date +%s)
coproc docker events --since "$since" --filter "container=$name" --filter event=health_status --format '{{.Action}}'
events=$!
t0=$EPOCHREALTIME
docker run -d --name "$name" --platform linux/arm64 \
  -e GCS_MIMIC_SYNC_DIR=/sync -v "$base/sync:/sync" "$image" >/dev/null || exit 1
while read -p line; do
  echo "  event: $line"
  [[ $line == *": healthy" ]] && break
done
t1=$EPOCHREALTIME
kill $events 2>/dev/null
printf 'start to healthy: %.2fs\n' $(( t1 - t0 ))

printf 'object 0500 changed\n' > "$base/sync/bench-bucket/file-0500.txt"
echo "changed: bench-bucket/file-0500.txt"
rescan_status=0
wait_status=0
t0=$EPOCHREALTIME
docker exec "$name" gcs-mimic sync rescan --timeout 60s >"$base/rescan.out" 2>&1 || rescan_status=$?
t1=$EPOCHREALTIME
docker exec "$name" gcs-mimic sync wait --timeout 60m >"$base/wait.out" 2>&1 || wait_status=$?
t2=$EPOCHREALTIME
printf 'rescan exit=%s %.2fs, wait exit=%s %.2fs, total %.2fs\n' \
  "$rescan_status" $(( t1 - t0 )) "$wait_status" $(( t2 - t1 )) $(( t2 - t0 ))
echo "rescan: $(cat "$base/rescan.out")"
echo "wait: $(cat "$base/wait.out")"

docker logs "$name" >"$base/server.log" 2>&1
docker rm -f "$name" >/dev/null
echo "server log lines: $(wc -l <"$base/server.log")"
echo "ingestions that took in a file after the start:"
grep '"event":"sync_ingested"' "$base/server.log" | grep -v '"ingested":0,' | tail -n +2
