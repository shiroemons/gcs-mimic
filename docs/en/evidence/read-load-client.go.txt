// Closed-loop HTTP load generator for comparing GCS emulators. Lives in the session
// scratchpad; it is deliberately not part of the repository.
package main

import (
	"bytes"
	"crypto/rand"
	"encoding/json"
	"flag"
	"fmt"
	"io"
	"net"
	"net/http"
	"os"
	"sort"
	"sync"
	"sync/atomic"
	"time"
)

func client(conns int) *http.Client {
	return &http.Client{
		Timeout: 60 * time.Second,
		Transport: &http.Transport{
			MaxIdleConns:        conns * 2,
			MaxIdleConnsPerHost: conns * 2,
			MaxConnsPerHost:     conns * 2,
			DisableCompression:  true,
			DialContext:         (&net.Dialer{Timeout: 5 * time.Second}).DialContext,
		},
	}
}

func seed(base, bucket string, objects, size, conc int) error {
	c := client(conc)
	body, _ := json.Marshal(map[string]string{"name": bucket})
	resp, err := c.Post(base+"/storage/v1/b?project=load", "application/json", bytes.NewReader(body))
	if err != nil {
		return err
	}
	io.Copy(io.Discard, resp.Body)
	resp.Body.Close()
	if resp.StatusCode >= 300 && resp.StatusCode != 409 {
		return fmt.Errorf("create bucket: %d", resp.StatusCode)
	}
	payload := make([]byte, size)
	rand.Read(payload)
	work := make(chan int)
	errs := make(chan error, conc)
	var wg sync.WaitGroup
	for w := 0; w < conc; w++ {
		wg.Add(1)
		go func() {
			defer wg.Done()
			for i := range work {
				url := fmt.Sprintf("%s/upload/storage/v1/b/%s/o?uploadType=media&name=img-%06d.jpg", base, bucket, i)
				resp, err := c.Post(url, "image/jpeg", bytes.NewReader(payload))
				if err != nil {
					select {
					case errs <- err:
					default:
					}
					continue
				}
				io.Copy(io.Discard, resp.Body)
				resp.Body.Close()
				if resp.StatusCode >= 300 {
					select {
					case errs <- fmt.Errorf("upload: %d", resp.StatusCode):
					default:
					}
				}
			}
		}()
	}
	for i := 0; i < objects; i++ {
		work <- i
	}
	close(work)
	wg.Wait()
	select {
	case e := <-errs:
		return e
	default:
		return nil
	}
}

var csv bool

func run(base, bucket, kind string, objects, size, conns int, d time.Duration) {
	c := client(conns)
	var next, errs, bytesRead atomic.Int64
	var wg sync.WaitGroup
	lat := make([][]time.Duration, conns)
	stop := time.Now().Add(d)
	start := time.Now()
	for w := 0; w < conns; w++ {
		wg.Add(1)
		go func(w int) {
			defer wg.Done()
			samples := make([]time.Duration, 0, 1<<16)
			for time.Now().Before(stop) {
				i := int(next.Add(1)) % objects
				var url string
				want := size
				switch kind {
				case "meta":
					url = fmt.Sprintf("%s/storage/v1/b/%s/o/img-%06d.jpg", base, bucket, i)
					want = -1
				case "media":
					url = fmt.Sprintf("%s/storage/v1/b/%s/o/img-%06d.jpg?alt=media", base, bucket, i)
				case "public":
					url = fmt.Sprintf("%s/%s/img-%06d.jpg", base, bucket, i)
				}
				t := time.Now()
				resp, err := c.Get(url)
				if err != nil {
					errs.Add(1)
					continue
				}
				n, err := io.Copy(io.Discard, resp.Body)
				resp.Body.Close()
				samples = append(samples, time.Since(t))
				bytesRead.Add(n)
				if err != nil || resp.StatusCode != 200 || (want >= 0 && int(n) != want) {
					errs.Add(1)
				}
			}
			lat[w] = samples
		}(w)
	}
	wg.Wait()
	elapsed := time.Since(start)
	all := make([]time.Duration, 0)
	for _, s := range lat {
		all = append(all, s...)
	}
	sort.Slice(all, func(i, j int) bool { return all[i] < all[j] })
	q := func(p float64) time.Duration {
		if len(all) == 0 {
			return 0
		}
		return all[int(float64(len(all)-1)*p)]
	}
	if csv {
		fmt.Printf("%s %d %.0f %d %d %d %d %d\n", kind, conns, float64(len(all))/elapsed.Seconds(),
			q(.50).Microseconds(), q(.95).Microseconds(), q(.99).Microseconds(), q(1).Microseconds(), errs.Load())
		return
	}
	fmt.Printf("%-7s c=%-3d rps=%8.0f MiB/s=%7.1f p50=%8s p95=%8s p99=%8s max=%8s n=%d errs=%d\n",
		kind, conns, float64(len(all))/elapsed.Seconds(), float64(bytesRead.Load())/elapsed.Seconds()/(1<<20),
		q(.50).Round(10*time.Microsecond), q(.95).Round(10*time.Microsecond), q(.99).Round(10*time.Microsecond), q(1).Round(10*time.Microsecond), len(all), errs.Load())
}

func main() {
	mode := flag.String("mode", "run", "seed or run")
	base := flag.String("base", "http://127.0.0.1:4443", "origin")
	bucket := flag.String("bucket", "load", "bucket")
	objects := flag.Int("objects", 1000, "object count")
	size := flag.Int("size", 100*1024, "object size in bytes")
	conns := flag.Int("c", 32, "connections")
	kind := flag.String("kind", "meta", "meta, media or public")
	d := flag.Duration("d", 10*time.Second, "duration")
	flag.BoolVar(&csv, "csv", false, "machine readable output")
	flag.Parse()
	if *mode == "seed" {
		t := time.Now()
		if err := seed(*base, *bucket, *objects, *size, 8); err != nil {
			fmt.Fprintln(os.Stderr, "seed failed:", err)
			os.Exit(1)
		}
		fmt.Printf("seeded %d objects of %d bytes in %s\n", *objects, *size, time.Since(t).Round(time.Millisecond))
		return
	}
	run(*base, *bucket, *kind, *objects, *size, *conns, *d)
}
